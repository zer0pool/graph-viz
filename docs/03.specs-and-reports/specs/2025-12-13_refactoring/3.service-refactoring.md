## GraphTraversalHelper

### Purpose

Extract and consolidate **BFS traversal logic** that's currently duplicated across multiple methods.

### Implementation

```python
# src/lineage_manager/services/helpers/graph_traversal_helper.py

from typing import Callable, Dict, Any, Optional, List, Set, Tuple
from lineage_manager.core.uow import GraphReadOnlyUnitOfWork


class GraphTraversalHelper:
    """Reusable graph traversal algorithms."""
    
    def __init__(self, uow: GraphReadOnlyUnitOfWork):
        self.uow = uow
    
    def bfs_traverse(
        self,
        start_node_id: int,
        start_node_type: str,
        direction: str = "both",
        max_depth: int = 3,
        node_filter: Optional[Callable] = None,
    ) -> Dict[str, Any]:
        """
        Generic BFS traversal for graph exploration.
        
        Args:
            start_node_id: Starting node database ID
            start_node_type: "job" or "table"
            direction: "upstream", "downstream", or "both"
            max_depth: Maximum traversal depth
            node_filter: Optional filter function for nodes
        
        Returns:
            {
                "nodes": {...},  # Dict of nodes by key
                "edges": [...],  # List of edges
                "visited": set(...)  # Set of visited (type, id) tuples
            }
        """
        visited = set([(start_node_type, start_node_id)])
        frontier = [(start_node_type, start_node_id)]
        nodes = {}
        edges = []
        
        for depth in range(max_depth):
            next_frontier = []
            
            for node_type, node_id in frontier:
                # Get neighbors based on node type and direction
                neighbors = self._get_neighbors(node_type, node_id, direction)
                
                for neighbor_type, neighbor_id, edge_type in neighbors:
                    if (neighbor_type, neighbor_id) in visited:
                        continue
                    
                    # Apply filter if provided
                    if node_filter and not node_filter(neighbor_type, neighbor_id):
                        continue
                    
                    visited.add((neighbor_type, neighbor_id))
                    next_frontier.append((neighbor_type, neighbor_id))
                    
                    # Add to result
                    self._add_node_to_result(nodes, neighbor_type, neighbor_id)
                    self._add_edge_to_result(edges, node_type, node_id, 
                                            neighbor_type, neighbor_id, edge_type)
            
            frontier = next_frontier
            if not frontier:
                break
        
        return {"nodes": nodes, "edges": edges, "visited": visited}
    
    def _get_neighbors(
        self, node_type: str, node_id: int, direction: str
    ) -> List[Tuple[str, int, str]]:
        """Get neighbors for a node based on type and direction."""
        neighbors = []
        
        if node_type == "job":
            if direction in ("upstream", "both"):
                # Get input tables
                tables = self.uow.job_table_links.get_tables_by_job_and_io_type(
                    node_id, "input"
                )
                neighbors.extend([("table", t.id, "input") for t in tables])
            
            if direction in ("downstream", "both"):
                # Get output tables
                tables = self.uow.job_table_links.get_tables_by_job_and_io_type(
                    node_id, "output"
                )
                neighbors.extend([("table", t.id, "output") for t in tables])
        
        elif node_type == "table":
            if direction in ("upstream", "both"):
                # Get producing jobs
                jobs = self.uow.job_table_links.get_jobs_by_table_and_io_type(
                    node_id, "output"
                )
                neighbors.extend([("job", j.id, "produces") for j in jobs])
            
            if direction in ("downstream", "both"):
                # Get consuming jobs
                jobs = self.uow.job_table_links.get_jobs_by_table_and_io_type(
                    node_id, "input"
                )
                neighbors.extend([("job", j.id, "consumes") for j in jobs])
        
        return neighbors
    
    def _add_node_to_result(self, nodes: dict, node_type: str, node_id: int):
        """Add node to result dictionary."""
        with self.uow:
            if node_type == "job":
                node = self.uow.jobs.get(node_id)
                if node:
                    nodes[f"job:{node.job_id}"] = {
                        "id": node.job_id,
                        "type": "job",
                        "name": node.name,
                        "owner": node.owner,
                        "enabled": node.enabled,
                    }
            elif node_type == "table":
                node = self.uow.tables.get_by_id(node_id)
                if node:
                    nodes[f"table:{node.full_name}"] = {
                        "id": node.full_name,
                        "type": "table",
                        "name": node.full_name,
                    }
    
    def _add_edge_to_result(
        self, edges: list, from_type: str, from_id: int,
        to_type: str, to_id: int, edge_type: str
    ):
        """Add edge to result list."""
        edges.append({
            "from": f"{from_type}:{from_id}",
            "to": f"{to_type}:{to_id}",
            "type": edge_type,
        })
    
    def ensure_node_entry(
        self, nodes: dict, node_type: str, obj
    ) -> str:
        """
        Ensure node dictionary entry exists.
        
        Returns:
            Canonical node key (e.g., "job:my_job" or "table:project.dataset.table")
        """
        if node_type == "job":
            key = f"job:{obj.job_id}"
            if key not in nodes:
                nodes[key] = {
                    "id": obj.job_id,
                    "type": "job",
                    "name": obj.name,
                    "owner": obj.owner,
                    "enabled": obj.enabled,
                }
        elif node_type == "table":
            key = f"table:{obj.full_name}"
            if key not in nodes:
                nodes[key] = {
                    "id": obj.full_name,
                    "type": "table",
                    "name": obj.full_name,
                }
        return key
    
    def extract_lineage_paths(
        self,
        start_node_id: int,
        start_node_type: str,
        direction: str,
        max_depth: int,
    ) -> List[List[str]]:
        """
        Extract all paths from start node.
        
        Returns:
            List of paths, where each path is a list of node keys
        """
        paths = []
        current_path = []
        
        def dfs(node_type: str, node_id: int, depth: int):
            if depth > max_depth:
                return
            
            current_path.append(f"{node_type}:{node_id}")
            
            neighbors = self._get_neighbors(node_type, node_id, direction)
            if not neighbors:
                # Leaf node - save path
                paths.append(list(current_path))
            else:
                for neighbor_type, neighbor_id, _ in neighbors:
                    dfs(neighbor_type, neighbor_id, depth + 1)
            
            current_path.pop()
        
        dfs(start_node_type, start_node_id, 0)
        return paths
```

**Methods**:
- `bfs_traverse()` - Generic BFS traversal (~60 lines)
- `ensure_node_entry()` - Node dict entry creation (~20 lines)
- `extract_lineage_paths()` - Path extraction (~40 lines)
- Helper methods (~80 lines total)

**Total**: ~200 lines

```python
# src/lineage_manager/services/graph_query_service.py

from typing import Dict, Any, Optional, List
from lineage_manager.core.uow import GraphReadOnlyUnitOfWork
from lineage_manager.services.helpers.graph_traversal_helper import GraphTraversalHelper


class GraphQueryService:
    """Read-only graph query operations."""
    
    def __init__(
        self,
        uow: GraphReadOnlyUnitOfWork,
        traversal_helper: GraphTraversalHelper,
    ):
        self.uow = uow
        self.traversal = traversal_helper
    
    def get_job(self, job_id: str) -> Optional[Dict[str, Any]]:
        """Get job with enriched metadata."""
        with self.uow:
            job = self.uow.jobs.get(job_id)
            if not job:
                return None
            
            return {
                "job_id": job.job_id,
                "name": job.name,
                "owner": job.owner,
                "enabled": job.enabled,
                "created_at": job.created_at.isoformat() if job.created_at else None,
                "updated_at": job.updated_at.isoformat() if job.updated_at else None,
            }
    
    def get_job_neighbors(
        self,
        job_id: str,
        level: int = 1,
        direction: str = "both",
        limit: Optional[int] = None,
    ) -> Dict[str, Any]:
        """
        Get neighbors around a job using BFS.
        
        Now uses GraphTraversalHelper - reduced from 126 lines to ~30 lines!
        """
        with self.uow:
            job = self.uow.jobs.get(job_id)
            if not job:
                return {
                    "status": "error",
                    "message": f"Job '{job_id}' not found"
                }
            
            # Use traversal helper
            result = self.traversal.bfs_traverse(
                start_node_id=job.id,
                start_node_type="job",
                direction=direction,
                max_depth=level,
            )
            
            return {
                "base_job": job_id,
                "nodes": list(result["nodes"].values()),
                "edges": result["edges"],
                "level": level,
                "direction": direction,
            }
    
    def get_table_neighbors(
        self,
        table_name: str,
        level: int = 1,
        direction: str = "both",
        limit: Optional[int] = None,
    ) -> Dict[str, Any]:
        """
        Get neighbors around a table using BFS.
        
        Now uses GraphTraversalHelper - reduced from 123 lines to ~30 lines!
        """
        with self.uow:
            table = self.uow.tables.get_by_full_name(table_name)
            if not table:
                return {
                    "status": "error",
                    "message": f"Table '{table_name}' not found"
                }
            
            # Use traversal helper
            result = self.traversal.bfs_traverse(
                start_node_id=table.id,
                start_node_type="table",
                direction=direction,
                max_depth=level,
            )
            
            return {
                "base_table": table_name,
                "nodes": list(result["nodes"].values()),
                "edges": result["edges"],
                "level": level,
                "direction": direction,
            }
    
    def get_table_dag(
        self,
        full_name: str,
        direction: str = "both",
        depth: int = 3,
        include_jobs: bool = True,
        include_tables: bool = True,
    ) -> Dict[str, Any]:
        """
        Get DAG for a table.
        
        Moved from TableRepository (was 139 lines) - now ~80 lines in service.
        """
        with self.uow:
            table = self.uow.tables.get_by_full_name(full_name)
            if not table:
                return {
                    "status": "error",
                    "message": f"Table '{full_name}' not found"
                }
            
            # Use traversal helper
            result = self.traversal.bfs_traverse(
                start_node_id=table.id,
                start_node_type="table",
                direction=direction,
                max_depth=depth,
                node_filter=lambda ntype, nid: (
                    (ntype == "job" and include_jobs) or
                    (ntype == "table" and include_tables)
                ),
            )
            
            return {
                "base_table": full_name,
                "nodes": list(result["nodes"].values()),
                "edges": result["edges"],
                "depth": depth,
                "direction": direction,
            }
    
    def get_table_lineage_hierarchy(
        self,
        table_name: str,
        max_depth: int = 3,
    ) -> Dict[str, Any]:
        """
        Get table lineage hierarchy.
        
        Now uses GraphTraversalHelper - reduced from 132 lines to ~70 lines!
        """
        with self.uow:
            table = self.uow.tables.get_by_full_name(table_name)
            if not table:
                return {"status": "error", "message": f"Table '{table_name}' not found"}
            
            # Get upstream paths
            upstream_paths = self.traversal.extract_lineage_paths(
                start_node_id=table.id,
                start_node_type="table",
                direction="upstream",
                max_depth=max_depth,
            )
            
            # Get downstream paths
            downstream_paths = self.traversal.extract_lineage_paths(
                start_node_id=table.id,
                start_node_type="table",
                direction="downstream",
                max_depth=max_depth,
            )
            
            return {
                "base_table": table_name,
                "upstream_paths": upstream_paths,
                "downstream_paths": downstream_paths,
                "max_depth": max_depth,
            }
    
    # ... other query methods (get_table_impact, get_table_triggers, etc.)
```

```python
# src/lineage_manager/services/graph_command_service.py

from typing import Dict, Any
from lineage_manager.core.uow import GraphUnitOfWork
from lineage_manager.schemas.job import JobRegister


class GraphCommandService:
    """Write operations for graph entities."""
    
    def __init__(self, uow: GraphUnitOfWork):
        self.uow = uow
    
    def register_job(self, job_data: JobRegister) -> str:
        """
        Register a new job.
        
        Split into helper methods - reduced from 102 lines to ~50 lines!
        """
        with self.uow:
            job = self._create_job_node(job_data)
            in_ids = self._process_reference_tables(job, job_data)
            out_ids = self._process_destination_table(job, job_data)
            self._create_upstream_relationships(job, in_ids)
            
            self.uow.commit()
            return job.job_id
    
    def _create_job_node(self, job_data: JobRegister):
        """Create job node (~20 lines)."""
        job = self.uow.jobs.get_or_create(
            job_id=job_data.job_id,
            name=job_data.name,
            owner=job_data.owner,
            enabled=job_data.enabled,
        )
        return job
    
    def _process_reference_tables(self, job, job_data) -> List[int]:
        """Process reference tables (~25 lines)."""
        in_ids = []
        for table_name in job_data.reference_tables or []:
            table = self.uow.tables.get_or_create(full_name=table_name)
            in_ids.append(table.id)
            
            # Create edge
            self.uow.edges.create_job_table_edge(
                job_id=job.id,
                table_id=table.id,
                io_type="input",
            )
        return in_ids
    
    def _process_destination_table(self, job, job_data) -> Optional[int]:
        """Process destination table (~15 lines)."""
        if not job_data.destination_table:
            return None
        
        table = self.uow.tables.get_or_create(
            full_name=job_data.destination_table
        )
        
        # Create edge
        self.uow.edges.create_job_table_edge(
            job_id=job.id,
            table_id=table.id,
            io_type="output",
        )
        return table.id
    
    def _create_upstream_relationships(self, job, in_ids: List[int]):
        """Create upstream job relationships (~20 lines)."""
        if not in_ids:
            return
        
        upstream_jobs = self.uow.jobs.find_upstream_jobs_by_output_tables(
            in_ids, job.id
        )
        
        for upstream_job in upstream_jobs:
            # Create job-to-job edge
            self.uow.edges.add(
                from_id=upstream_job.id,
                to_id=job.id,
                edge_type="job_dependency",
            )
            
            # Update closure table
            self.uow.closures.add_direct(upstream_job.id, job.id)
            self.uow.closures.expand_closure(upstream_job.id, job.id)
    
    def toggle_job_enabled(self, job_id: str) -> Dict[str, Any]:
        """Toggle job enabled status."""
        with self.uow:
            job = self.uow.jobs.get(job_id)
            if not job:
                return {"status": "error", "message": f"Job '{job_id}' not found"}
            
            job.enabled = not job.enabled
            self.uow.commit()
            
            return {
                "status": "success",
                "job_id": job_id,
                "enabled": job.enabled,
            }
    
    # ... other mutation methods (set_table_trigger, reset_graph, etc.)
```

```python
from typing import Dict, Any, List
from lineage_manager.core.uow import GraphUnitOfWork
from lineage_manager.adapters.job_manager_adapter import JobManagerPort
from lineage_manager.services.graph_command_service import GraphCommandService
from lineage_manager.schemas.lineage import SchedulingLineage


class GraphSyncService:
    """Sync operations with external Job Manager."""
    
    def __init__(
        self,
        uow: GraphUnitOfWork,
        job_manager: JobManagerPort,
        command_service: GraphCommandService,
    ):
        self.uow = uow
        self.job_manager = job_manager
        self.command = command_service
    
    def sync_from_lineage(self, lineage: SchedulingLineage) -> None:
        """
        Sync a single lineage to graph.
        """
        with self.uow:
            # Extract job properties
            job_props = self._extract_job_properties(lineage)
            
            # Create or update node
            node = self._create_or_update_node("job", lineage.job_id, job_props)
            
            # Process references and destinations
            self._process_lineage_tables(node, lineage)
            
            self.uow.commit()
    
    def sync_single_job(
        self, lineage: SchedulingLineage
    ) -> Dict[str, Any]:
        """
        Sync single job with validation.
        
        Split into helpers - reduced from 100 lines to ~50 lines!
        """
        self._validate_lineage(lineage)
        result = self._perform_sync(lineage)
        return result

        """Validate lineage data (~15 lines)."""
        if not lineage.job_id:
            raise ValueError("job_id is required")
        if not lineage.job_name:
            raise ValueError("job_name is required")

            self._process_lineage_tables(node, lineage)
            self.uow.commit()
            
            return {
                "status": "success",
                "job_id": lineage.job_id,
                "synced_at": datetime.utcnow().isoformat(),
            }

            "name": lineage.job_name,
            "owner": lineage.owner,
            "enabled": lineage.enabled,
            "metadata": {
                "schedule": lineage.schedule,
                "tags": lineage.tags,
            },
        }
    
    # ... other sync methods
```
